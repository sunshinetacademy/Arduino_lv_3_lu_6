✅ الإجابة:

عبر حساسين يعملان معاً داخل شريحة واحدة:

1. مقياس التسارع (Accelerometer): يقيس قوة التسارع والجاذبية على 3 محاور. عندما تميل الجهاز، تتغير قراءات الجاذبية — فيعرف أنه مائل.

2. الجيروسكوب (Gyroscope): يقيس سرعة الدوران على 3 محاور. يعرف الجهاز "كم يدور في الثانية".

معاً = 6 درجات حرية (6 DOF) — فهم كامل لحركة الجهاز في الفضاء!

🔬 ما هو MPU6050؟

03 / 11

💡 حساسان في شريحة

MPU6050 يحوي حساسين متقدمين:

⚖️ مقياس التسارع (Accelerometer)

يقيس التسارع على X, Y, Z بالـ g. الجاذبية الأرضية = 1g.

🌀 الجيروسكوب (Gyroscope)

يقيس سرعة الدوران على X, Y, Z بالدرجة/ثانية.

🎯 6 درجات حرية (6 DOF)

🎯

3 محاور للتسارع + 3 محاور للدوران

X

يمين / يسار

Y

أمام / خلف

Z

أعلى / أسفل

🎯 المصطلحات الثلاثة المهمة:
Pitch (ميل للأمام/خلف)
الدوران حول المحور X
Roll (ميل يمين/يسار)
الدوران حول المحور Y
Yaw (لف حول النفس)
الدوران حول المحور Z

⚠️ تحذير حرج

🛡️ 3 قواعد مهمة

05 / 11
⚡ تحذير 1 — الجهد:

MPU6050 يعمل بـ 3.3V. توصيله بـ 5V يُتلفه فوراً — نفس القاعدة الذهبية. بعض لوحات GY-521 تحتوي منظم جهد داخلي يقبل 5V لكن لا تعتمد على ذلك.

⚠️ تحذير 2 — حساسية MEMS:

MPU6050 يحتوي أجزاء ميكروية دقيقة تتحرك داخلياً. لا تسقطه ولا تعرضه لصدمات — قد يتحطم الجزء الداخلي.

⚠️ تحذير 3 — المعايرة في مكان ثابت:

عند تشغيل المشروع، اترك الحساس ثابتاً على الطاولة لحظة البداية. بعض المكتبات تعاير عند الإقلاع — الحركة تُفسد المعايرة.

💡 نصيحة عملية: MPU6050 له ضجيج (Noise) طبيعي في القراءات — حوالي ±0.05g. هذا ليس عيباً — بل طبيعي. الحل: استخدام مرشح (Filter) بسيط مثل المتوسط المتحرك.

نتأكد أن MPU6050 يعمل — ونقرأ المحاور الستة على Serial Monitor.

#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

Adafruit_MPU6050 mpu;

void setup() {
  Serial.begin(115200);

  if (!mpu.begin()) {
    Serial.println("MPU6050 not found!");
    while (1);
  }

  // ضبط المدى
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  Serial.println("MPU6050 Ready!");
  delay(1000);
}

void loop() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  Serial.print("Accel X: ");
  Serial.print(a.acceleration.x);
  Serial.print(" Y: ");
  Serial.print(a.acceleration.y);
  Serial.print(" Z: ");
  Serial.println(a.acceleration.z);

  Serial.print("Gyro  X: ");
  Serial.print(g.gyro.x);
  Serial.print(" Y: ");
  Serial.print(g.gyro.y);
  Serial.print(" Z: ");
  Serial.println(g.gyro.z);

  Serial.println("---");
  delay(500);
}
✅ مستوي على الطاولة: Accel Z ≈ 9.8، X و Y ≈ 0
❌ لو ظهر "not found": تحقق من التوصيل + العنوان 0x68

📐 الكود 2: مقياس الميلان على OLED

#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_W 128
#define SCREEN_H 64
#define OLED_ADDR 0x3C

Adafruit_MPU6050 mpu;
Adafruit_SSD1306 display(SCREEN_W, SCREEN_H, &Wire, -1);

void setup() {
  Serial.begin(115200);

  if (!mpu.begin()) {
    Serial.println("MPU6050 failed!");
    while (1);
  }
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);

  display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("MPU6050 Ready");
  display.display();
  delay(1000);
}

void loop() {
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  // حساب الميلان بالمعادلة الهندسية
  float pitch = atan2(a.acceleration.y, a.acceleration.z) * 180 / PI;
  float roll  = atan2(-a.acceleration.x, a.acceleration.z) * 180 / PI;

  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("== Motion Meter ==");

  // Pitch
  display.setCursor(0, 14);
  display.print("Pitch: ");
  display.print(pitch, 1);
  display.println(" deg");

  // Roll
  display.setCursor(0, 26);
  display.print("Roll:  ");
  display.print(roll, 1);
  display.println(" deg");

  // جيروسكوب Z
  display.setCursor(0, 38);
  display.print("GyroZ: ");
  display.print(g.gyro.z, 1);
  display.println(" d/s");

  // الحرارة
  display.setCursor(0, 52);
  display.print("Temp:  ");
  display.print(temp.temperature, 1);
  display.println(" C");

  display.display();
  delay(200);
}
🎯 3 أفكار مفتاحية:

mpu.getEvent(&a, &g, &temp) — يقرأ كل البيانات دفعة واحدة.
atan2(y, z) — دالة رياضية لحساب الزاوية من الإحداثيات.
* 180 / PI — التحويل من راديان إلى درجات.
